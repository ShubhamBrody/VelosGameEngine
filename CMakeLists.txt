cmake_minimum_required(VERSION 3.28)
project(Velos VERSION 0.1.0 LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native Velos preview currently targets Windows x64.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

include(cmake/Dependencies.cmake)

add_library(velos_core INTERFACE)
target_include_directories(velos_core INTERFACE "${PROJECT_SOURCE_DIR}/engine")
target_compile_features(velos_core INTERFACE cxx_std_20)
if(MSVC)
    target_compile_options(velos_core INTERFACE /W4 /WX /permissive- /utf-8 /Zc:__cplusplus)
    target_compile_definitions(velos_core INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
endif()

add_library(velos_scene STATIC engine/scene/Scene.cpp engine/scene/History.cpp)
target_link_libraries(velos_scene PUBLIC velos_core EnTT::EnTT nlohmann_json::nlohmann_json)

add_library(velos_platform STATIC engine/platform/Files.cpp engine/platform/Window.cpp)
target_link_libraries(velos_platform PUBLIC velos_core bcrypt shell32 ole32)
add_library(velos_cache STATIC engine/assets/DiskCache.cpp)
target_link_libraries(velos_cache PUBLIC velos_platform)

add_library(velos_mesh STATIC engine/assets/Mesh.cpp)
target_include_directories(velos_mesh SYSTEM PRIVATE "${cgltf_SOURCE_DIR}")
target_link_libraries(velos_mesh PUBLIC velos_cache PRIVATE meshoptimizer)

add_library(velos_texture STATIC engine/assets/Texture.cpp)
target_link_libraries(velos_texture PUBLIC velos_cache DirectXTex)

add_library(velos_renderer STATIC engine/rhi/d3d12/Renderer.cpp)
add_library(velos_render_plan STATIC engine/render/DrawBatches.cpp)
target_link_libraries(velos_render_plan PUBLIC velos_core)
target_link_libraries(velos_renderer PUBLIC velos_mesh velos_texture PRIVATE velos_render_plan velos_imgui d3d12 dxgi dxguid windowscodecs)

add_library(velos_bridge STATIC engine/bridge/Extract.cpp)
target_link_libraries(velos_bridge PUBLIC velos_scene)
add_library(velos_physics STATIC engine/physics/PhysicsWorld.cpp)
target_link_libraries(velos_physics PUBLIC velos_scene PRIVATE Jolt)
add_library(velos_ai STATIC engine/ai/StreamDecoder.cpp engine/ai/Http.cpp engine/ai/Assistant.cpp)
target_link_libraries(velos_ai PUBLIC velos_cache nlohmann_json::nlohmann_json PRIVATE winhttp advapi32)
add_library(velos_package STATIC engine/assets/Package.cpp)
target_link_libraries(velos_package PUBLIC velos_scene velos_platform)
add_executable(VelosEditor editor/main.cpp editor/Editor.cpp editor/AssistantPanel.cpp)
target_link_libraries(VelosEditor PRIVATE velos_renderer velos_bridge velos_gizmo velos_physics velos_ai velos_package)
add_executable(VelosRuntime runtime/main.cpp)
target_link_libraries(VelosRuntime PRIVATE velos_renderer velos_bridge velos_physics velos_package)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT VelosEditor)

file(GLOB VELOS_DXC_CANDIDATES "C:/Program Files (x86)/Windows Kits/10/bin/*/x64/dxcompiler.dll")
list(SORT VELOS_DXC_CANDIDATES COMPARE NATURAL ORDER DESCENDING)
if(NOT VELOS_DXC_CANDIDATES)
    message(FATAL_ERROR "Install the Windows SDK including DXC (dxcompiler.dll).")
endif()
list(GET VELOS_DXC_CANDIDATES 0 VELOS_DXC_DLL)
get_filename_component(VELOS_DXC_DIR "${VELOS_DXC_DLL}" DIRECTORY)
if(CMAKE_CONFIGURATION_TYPES)
    set(VELOS_RUNTIME_DIR "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
else()
    set(VELOS_RUNTIME_DIR "${CMAKE_CURRENT_BINARY_DIR}")
endif()
set(VELOS_RUNTIME_COPY_COMMANDS
    COMMAND ${CMAKE_COMMAND} -E make_directory "${VELOS_RUNTIME_DIR}/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${VELOS_DXC_DLL}" "${VELOS_RUNTIME_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${VELOS_DXC_DIR}/dxil.dll" "${VELOS_RUNTIME_DIR}"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${PROJECT_SOURCE_DIR}/shaders" "${VELOS_RUNTIME_DIR}/shaders")
foreach(dependency IN ITEMS imgui imguizmo entt json cgltf jolt directxtex meshoptimizer)
    file(GLOB license_files "${${dependency}_SOURCE_DIR}/LICENSE*")
    foreach(license_file IN LISTS license_files)
        if(NOT IS_DIRECTORY "${license_file}")
            get_filename_component(license_name "${license_file}" NAME)
            list(APPEND VELOS_RUNTIME_COPY_COMMANDS COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${license_file}" "${VELOS_RUNTIME_DIR}/licenses/${dependency}-${license_name}")
        endif()
    endforeach()
endforeach()
add_custom_target(velos_runtime_data ALL ${VELOS_RUNTIME_COPY_COMMANDS} VERBATIM)
add_dependencies(VelosEditor velos_runtime_data)
add_dependencies(VelosRuntime velos_runtime_data)

add_executable(velos_ai_probe tools/ai_probe.cpp)
target_link_libraries(velos_ai_probe PRIVATE velos_ai)
add_executable(velos_graphics_fixture tools/graphics_fixture.cpp)
target_link_libraries(velos_graphics_fixture PRIVATE velos_scene velos_texture)
